## 디렉토리 구조 (목표)

```
src/
├── App.tsx                    # React Router + Layout

├── index.css

│   └── youth.ts
├── components/
│   ├── layout/
│   │   ├── Nav.tsx
│   │   ├── Footer.tsx
│   │   └── Layout.tsx         # Nav + Outlet + Footer
│   ├── shared/

│   │   ├── Loader.tsx

│   │   ├── AthleteCard.tsx    # 선수 카드 (레전드/유스 공용)
│   │   └── FilterTabs.tsx     # 필터 탭 공통
│   └── home/                  # 메인페이지 전용
│       ├── Hero.tsx + WebGLHero.tsx
│       ├── Manifesto.tsx
│       ├── MediaPreview.tsx   # (← HorizontalGallery)
│       ├── LegendHighlight.tsx # (← StageSections ON)
│       ├── YouthPreview.tsx   # (← StageSections OFF)
│       ├── LegendPreview.tsx  # (← HallOfFame)
│       ├── StoreSection.tsx
│       ├── Partners.tsx
│       └── Socials.tsx
└── pages/
    ├── HomePage.tsx
    ├── LegendsPage.tsx
    ├── LegendDetailPage.tsx
    ├── MediaPage.tsx
    ├── YouthPage.tsx
    └── AboutPage.tsx
```
